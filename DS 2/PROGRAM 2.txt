/*Write a program to find out the factorial of a number using recursion
(stack)*\

#include <stdio.h>

int factorial(int n)
{
    if(n == 0 || n == 1)
    {
        return 1;
    }
    else
    {
        return n * factorial(n - 1);
    }
}

int main()
{
    int n;

    printf("Enter a number: ");
    scanf("%d", &n);

    if(n < 0)
    {
        printf("Factorial does not exist for negative numbers.");
    }
    else
    {
        printf("Factorial of %d = %d", n, factorial(n));
    }

    return 0;
}
