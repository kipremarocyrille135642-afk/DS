/*Write a program to find the factorial of a given integer number using
stack.*\

#include <stdio.h>
#define MAX 100

int stack[MAX];
int top = -1;

void push(int x)
{
    if(top < MAX - 1)
    {
        stack[++top] = x;
    }
}

int pop()
{
    if(top == -1)
    {
        return -1;
    }
    return stack[top--];
}

int main()
{
    int n, i, fact = 1;

    printf("Enter a number: ");
    scanf("%d", &n);

    if(n < 0)
    {
        printf("Factorial does not exist for negative numbers.");
    }
    else
    {
        for(i = 1; i <= n; i++)
        {
            push(i);
        }

        while(top != -1)
        {
            fact = fact * pop();
        }

        printf("Factorial of %d = %d", n, fact);
    }

    return 0;
}
